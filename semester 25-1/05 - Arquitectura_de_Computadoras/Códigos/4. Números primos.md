;Practica3---------------------------------------------------------------------
;Generar los primeros numeros primos hasta llenar 16 bits
;El primero se guarda en la direccion 0x800, el segundo en 0x802
;y asi sucesivamente
    
MOV #65535, W5 ;Limite para encontrar numeros primos
MOV #2, W2 ;Posible
MOV #2, W3 ;Divisor
MOV #0X800, W7 ;La primera direccion donde se guardara el numero primo

cabeza:    
CPSGT W2, W5 ;Si el candidato a primo es mayor al limite, termina el proceso
BRA terminar
    
REPEAT #17   ;Divide posible / divisor
DIV.U  W2,W3 ;Guarda el residuo en W1
   
CP W1, #0 ;Compara el residuo de la division con 0
BRA Z, preguardar    ; Si el residuo es o salta a preguardar

INC W3, W3 ;Si el residuo no fue 0 entonces incrementa en 1 el divisor

BRA cabeza ;Regresamos a cabeza para verificar el siguiente divisor
    
preguardar:
 
CP W2, W3 ;Compara W2 y W3 para saber si estamos dividiendo entre el mismo
BRA Z, guardar ;Si es el mismo numero saltamos a guardar el numero primo 
    
sigprimo:
    
MOV #2, W3
;Si el numero no es el mismo que el divisor, al tener residuo 0, no es primo
INC W2, W2  ;Pasamos a probar con el siguiente candidato a primo
    
BRA cabeza ;Regresamos a cabeza para verificar el siguiente candidato

guardar:
MOV W2, [W7]
;Guarda el número primo en la dirección de memoria 0x800, 0x802, ...
ADD #2, W7
;Incrementa la dirección en 2 bytes para la siguiente posición
    
BRA sigprimo  ;Seguimos para verificar el siguiente candidato

terminar:
    NOP
